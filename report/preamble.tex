% XeLaTeX or LuaLaTeX only. This document sets long s (U+017F) and the
% correction pass's own markers (U+27E6, U+27E7) as literal characters, and
% pdfLaTeX cannot without a workaround per character. Libertinus is the font
% because it is the one candidate that covers both: DejaVu Serif and Noto Serif
% were probed on 2026-09-18 and neither has U+27E6 or U+27E7.

% Everything from here to the quotation macros styles an `article`, and
% beamer already owns all of it: it loads geometry with its own options, so
% loading it again with ours is an option clash that stops the deck's build
% outright, and titlesec has no meaning for a frame. The deck still shares
% the font, the language setup and the macros below the guard, which is the
% point of one preamble.
%
% \makeatletter sits outside the guard on purpose. TeX tokenises the whole
% branch as it reads it, so `@` has to be a letter before the argument is
% read, not once the branch is taken; otherwise \@title inside tokenises as
% \@ followed by `title' and crashes exactly as the note below describes.
\makeatletter
\@ifclassloaded{beamer}{}{%
\usepackage[a4paper,top=2.5cm,bottom=2.5cm,left=2.5cm,right=2.5cm]{geometry}

% Float placement. At LaTeX's defaults this report deferred every figure and
% table past the bibliography: \topfraction 0.7 and \totalnumber 3 are too
% tight for a ten-page document that carries eight floats, so each one failed
% its page and queued. Loosened here rather than by scattering [!ht] over the
% sections, because the sections should say what a figure is, not where it goes.
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.6}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.72}
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}

% A plain title page in the same idiom as the template's \notitlepagelayout:
% a ruled, uppercase, coloured title block, author and date below it. Called
% from main.tex after \title/\author/\date, in place of the class's
% \maketitle.
%
% \@title/\@author/\@date need catcode 11 on `@`. That catcode is set inside
% every .cls/.sty while it loads, but this file reaches TeX through a plain
% \input, which does not touch it, so `@` sits at its usual catcode 12
% throughout: \@title then tokenises as \@ (the kernel's sentence-spacing
% primitive) followed by four ordinary letters "title", and \@ run in
% vertical mode is the "\spacefactor in vertical mode" crash this hunted
% down on 2026-09-22 (it first looked like a \MakeUppercase/fontspec bug).
\makeatletter
\newcommand{\reportmaketitle}{%
  \begin{titlepage}
    \vspace*{4cm}
    \begin{center}
      {\color{reportblue}\rule{\textwidth}{1pt}}\\[0.4cm]
      {\color{reportblue}\Huge\scshape \MakeUppercase{\@title}}\\[0.4cm]
      {\color{reportblue}\rule{\textwidth}{1pt}}\\[3cm]
      {\Large \@author}\\[0.5cm]
      {\large \@date}
    \end{center}
    \vfill
  \end{titlepage}
}
\makeatother

% The abstract is a page of its own: the title page, then the abstract, then
% section 1. `article` would otherwise set it at the head of the first body
% page, where it reads as a preamble to the introduction rather than as the
% thing a reader may stop at.
\renewenvironment{abstract}{%
  \clearpage
  \vspace*{2cm}
  \begin{center}{\color{reportblue}\large\bfseries\sffamily\MakeUppercase{\abstractname}}\end{center}
  \vspace{0.4cm}
  \begin{quotation}\noindent\ignorespaces
}{%
  \end{quotation}\clearpage
}

% A claim in section 1. \noindent on every one of them, because the three were
% ordinary paragraphs and only the first sat flush left: the other two took the
% paragraph indent and the labels did not line up.
\newcommand{\claim}[2]{%
  \par\medskip\noindent\textbf{#1}\ \textbf{#2}\ \ignorespaces
}
}
\makeatother

% A file name in this repository.
%
% \detokenize because a path carries underscores, and an underscore reaching
% \texttt at its usual catcode is a subscript: src/amadis_htr/ fails the build
% with "Missing $ inserted". Detokenising sets every character to catcode 12,
% so the path prints as written.
%
% url's \path was tried twice and reverted both times. Narrowing \UrlBreaks
% does not stop TeX breaking after an explicit hyphen, and widening it to every
% letter makes url insert a penalty at each one, which renders the path
% letter-spaced: "d a t a / l o c a l i s a t i o n". A path that runs long is
% fixed by rewording the sentence around it.
\newcommand{\file}[1]{\texttt{\detokenize{#1}}}

% A long path or DOI in a narrow measure would otherwise reach the margin.
% Stretching the line a little is the cheaper of the two.
\setlength{\emergencystretch}{1.5em}
