\begin{frame}{The problem}
  \begin{itemize}
    \item \ocrBooks{} books, \ocrPages{} pages of 16th-century French romance
    \item One year of pipeline, one recognition figure: a validation CER on a
      10\% split of one volume
    \item No figure for \emph{Amadis de Gaule}, no WER, no baseline, no
      evaluation of either LLM pass
  \end{itemize}
  \vfill
  This is the measurement.
\end{frame}

\begin{frame}{What got measured, and what did not}
  \begin{itemize}
    \item \textbf{E2} correction --- the accept rule, on \corrOffered{} lines
      from a seeded \corrPages{}-page sample
    \item \textbf{E5} localisation --- \locPiecesWorkbook{} pieces under the
      pre-registered gate, plus \locPiecesCatalogue{} imported later and
      reported beside them, never pooled
    \item \textbf{E6} throughput and structure --- \ocrPages{} pages
    \item \textbf{E1, E3, E4} --- blocked on a 20-page annotation pass
  \end{itemize}
  \vfill
  Of the five experiments that test a claim, three are unmeasured.
\end{frame}

\begin{frame}{Limitations}
  \begin{itemize}
    \item \textbf{C1 is unevaluated.} No gold set, so no recognition figure.
    \item The structural rates are hand-checked, and named for it.
    \item Five of six thresholds have never been swept.
    \item E2 measures refusal, not improvement.
    \item \textbf{The accept rule has a hole}, and
      \corrFindFallbackShare{} of E2's lines sat on it --- a span-locating
      fallback that can mark one word twice, which the rule cannot detect. C2
      rests on under a third of its own evidence.
  \end{itemize}
\end{frame}

\begin{frame}{What a successor does first}
  \begin{enumerate}
    \item Annotate the 20-page gold set. Ten hours, not a research problem.
    \item Add a second annotator, for a real inter-annotator floor.
    \item Fix the span-locating fallback in the pipeline.
  \end{enumerate}

  Everything blocked in this report is blocked behind the first line.
\end{frame}
